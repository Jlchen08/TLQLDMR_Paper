\begin{tikzpicture}[

% ============ ROW 1: INPUT DATA ============
\node[groupbox=inputborder, minimum width=10cm, minimum height=1.8cm] (inputgroup) {};
\node[above=-1pt of inputgroup.north, font=\footnotesize\bfseries, color=inputborder] {Data Input Layer};

\node[databox=inputborder] (source) at ($(inputgroup.center)+(-2.5,0)$) {Source Domain $D_S$\\Normal Weather\\($n_S$ samples)};
\node[databox=inputborder] (target) at ($(inputgroup.center)+(2.5,0)$) {Target Domain $D_T$\\Extreme Weather\\($n_T \ll n_S$)};

\node[font=\tiny, color=inputborder] at ($(source.east)!0.5!(target.west)$) {Distribution Shift};

% ============ ROW 2: KERNEL MAPPING ============
\node[groupbox=kernelborder, minimum width=10cm, minimum height=1.5cm, below=0.8cm of inputgroup] (kernelgroup) {};
\node[above=-1pt of kernelgroup.north, font=\footnotesize\bfseries, color=kernelborder] {Feature Mapping};

\node[databox=kernelborder, minimum width=4cm] (kernel) at ($(kernelgroup.center)+(-1.8,0)$) {Kernel $\phi: \mathcal{X} \to \mathcal{H}$\\(RBF / Nystr\"om)};
\node[smallbox=kernelborder, minimum width=3.2cm] (complexity) at ($(kernelgroup.center)+(2.5,0)$) {Complexity: $O(N^3)\!\to\!O(TNm)$};

% ============ ROW 3: OPTIMIZATION ============
\node[groupbox=optborder, minimum width=10cm, minimum height=3.2cm, below=0.8cm of kernelgroup] (optgroup) {};
\node[above=-1pt of optgroup.north, font=\footnotesize\bfseries, color=optborder] {Joint Optimization $\min_{\mathbf{w},b} J(\mathbf{w},b)$};

% 2x2 grid of losses
\node[lossbox=srborder] (sr) at ($(optgroup.center)+(-2.3,0.7)$) {Structural Risk\\$\frac{1}{2}\|\mathbf{w}\|^2$};
\node[lossbox=varborder] (var) at ($(optgroup.center)+(2.3,0.7)$) {Variance (LDMR)\\$\lambda_1\!\sum\!(u_i\!-\!\bar{u})^2$};
\node[lossbox=mmdborder] (mmd) at ($(optgroup.center)+(-2.3,-0.7)$) {MMD Adaptation\\$\lambda_2\text{MMD}^2(D_S,D_T)$};
\node[lossbox=lossborder] (loss) at ($(optgroup.center)+(2.3,-0.7)$) {Pinball Loss\\$C_S\!\sum\!L_\tau\!+\!C_T\!\sum\!L_\tau$};

% Plus signs
\node[font=\normalsize\bfseries, color=gray!60] at ($(sr.east)!0.5!(var.west)$) {+};
\node[font=\normalsize\bfseries, color=gray!60] at ($(mmd.east)!0.5!(loss.west)$) {+};
\node[font=\normalsize\bfseries, color=gray!60] at ($(sr.south)!0.5!(mmd.north)$) {+};
\node[font=\normalsize\bfseries, color=gray!60] at ($(var.south)!0.5!(loss.north)$) {+};

\node[databox=outputborder, minimum width=3.5cm] (model) at ($(outputgroup.center)+(-2.2,0)$) {Optimal Model\\$f(\mathbf{x})\!=\!\mathbf{w}^T\phi(\mathbf{x})\!+\!b$};

\node[smallbox=outputborder] (q95) at ($(outputgroup.center)+(2,0.35)$) {Upper: $\hat{q}_{0.95}$};
\node[smallbox=outputborder] (q50) at ($(outputgroup.center)+(2,0)$) {Median: $\hat{q}_{0.50}$};
\node[smallbox=outputborder] (q05) at ($(outputgroup.center)+(2,-0.35)$) {Lower: $\hat{q}_{0.05}$};
